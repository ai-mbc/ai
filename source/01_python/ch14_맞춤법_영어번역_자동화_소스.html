from pathlib import Path
import time

from selenium import webdriver
from selenium.common.exceptions import (
    NoSuchElementException,
    StaleElementReferenceException,
)
from selenium.webdriver.common.by import By
from selenium.webdriver.common.keys import Keys
from selenium.webdriver.support.ui import WebDriverWait


# -----------------------------------------------------------------------------
# 1. 파일 경로와 처리 단위
# -----------------------------------------------------------------------------
DATA_DIR = Path(r"C:\ai\source\01_python\data")

INPUT_PATH = DATA_DIR / "ch14_입력_3천자 이상.txt"
KOREAN_OUTPUT_PATH = DATA_DIR / "ch14_맞춤법후.txt"

# 사용자가 지정한 파일명(‘번영본’)을 그대로 사용한다.
ENGLISH_OUTPUT_PATH = DATA_DIR / "ch14_자동화영어번영본.txt"

# 전체 작업과 Papago 번역은 3,000자 단위로 처리한다.
WORK_BLOCK_SIZE = 3000
PAPAGO_BLOCK_SIZE = 3000

# 네이버 맞춤법 검사기는 공식적으로 한 번에 300자까지만 지원한다.
# 따라서 각 3,000자 작업 블록을 다시 300자 이하로 나누어 전송한다.
NAVER_REQUEST_SIZE = 300

WAIT_SECONDS = 25
REQUEST_PAUSE_SECONDS = 1.2


# -----------------------------------------------------------------------------
# 2. 현재 화면과 교재 화면을 함께 지원하는 선택자
# -----------------------------------------------------------------------------
NAVER_INPUT_LOCATORS = (
    (By.CSS_SELECTOR, "textarea.txt_gray"),
    (By.CLASS_NAME, "txt_gray"),
)
NAVER_BUTTON_LOCATORS = (
    (By.CSS_SELECTOR, "button.btn_check"),
    (By.CLASS_NAME, "btn_check"),
)
NAVER_RESULT_LOCATORS = (
    (By.CSS_SELECTOR, "p._result_text.stand_txt"),
    (By.CSS_SELECTOR, "._result_text.stand_txt"),
)

PAPAGO_SOURCE_LOCATORS = (
    (By.CSS_SELECTOR, "[data-testid='source-editor']"),
    (By.CSS_SELECTOR, "div[contenteditable='true'][role='textbox']"),
    (By.ID, "txtSource"),
)
PAPAGO_TARGET_LOCATORS = (
    (By.CSS_SELECTOR, "[data-testid='target-editor']"),
    (By.CSS_SELECTOR, "div[contenteditable='false'][role='textbox']"),
    (By.ID, "txtTarget"),
)


# -----------------------------------------------------------------------------
# 3. 공통 함수
# -----------------------------------------------------------------------------
def split_by_sentence(text, max_chars):
    """문장 끝을 우선하여 max_chars 이하의 문자열 목록으로 나눈다."""
    remaining = text.strip()
    chunks = []

    while len(remaining) > max_chars:
        temp = remaining[:max_chars]
        cut = max(temp.rfind(mark) for mark in (".", "?", "!", "\n"))

        # 문장부호가 너무 앞에 있거나 없으면 마지막 공백을 사용한다.
        if cut < max_chars // 2:
            cut = temp.rfind(" ")

        # 공백도 없으면 데이터 손실 없이 max_chars에서 강제로 나눈다.
        if cut == -1:
            cut = max_chars - 1

        chunk = remaining[: cut + 1].strip()
        if chunk:
            chunks.append(chunk)
        remaining = remaining[cut + 1 :].lstrip()

    if remaining:
        chunks.append(remaining)

    return chunks


def locate_visible(driver, locators):
    """후보 선택자 중 화면에 표시된 첫 번째 요소를 반환한다."""
    for by, value in locators:
        try:
            element = driver.find_element(by, value)
            if element.is_displayed():
                return element
        except (NoSuchElementException, StaleElementReferenceException):
            pass
    return None


def wait_visible(driver, locators):
    return WebDriverWait(driver, WAIT_SECONDS).until(
        lambda current_driver: locate_visible(current_driver, locators)
    )


def read_visible_text(driver, locators):
    element = locate_visible(driver, locators)
    if element is None:
        return ""

    try:
        return element.text.strip()
    except StaleElementReferenceException:
        return ""


# -----------------------------------------------------------------------------
# 4. 네이버 맞춤법 검사
# -----------------------------------------------------------------------------
def open_naver_spell_checker(driver):
    url = (
        "https://search.naver.com/search.naver?"
        "query=%EB%A7%9E%EC%B6%A4%EB%B2%95+%EA%B2%80%EC%82%AC%EA%B8%B0"
    )
    driver.get(url)
    wait_visible(driver, NAVER_INPUT_LOCATORS)


def spellcheck_one_chunk(driver, text):
    if len(text) > NAVER_REQUEST_SIZE:
        raise ValueError(f"네이버 전송 문장이 300자를 초과했습니다: {len(text)}자")

    textarea = wait_visible(driver, NAVER_INPUT_LOCATORS)
    textarea.clear()
    textarea.send_keys(text)

    button = wait_visible(driver, NAVER_BUTTON_LOCATORS)
    button.click()

    # 결과가 이전 화면에서 바뀔 시간을 확보한 뒤 실제 텍스트를 검증한다.
    time.sleep(REQUEST_PAUSE_SECONDS)

    def corrected_text(current_driver):
        result = read_visible_text(current_driver, NAVER_RESULT_LOCATORS)
        if not result:
            return False
        if "입력해 주세요" in result:
            return False
        return result

    return WebDriverWait(driver, WAIT_SECONDS).until(corrected_text)


def spellcheck_full_text(driver, original_text):
    # 사용자가 요청한 3,000자 기준의 상위 작업 블록
    work_blocks = split_by_sentence(original_text, WORK_BLOCK_SIZE)

    # 네이버의 실제 300자 제한을 지키기 위한 하위 전송 블록
    naver_chunks = []
    for block in work_blocks:
        naver_chunks.extend(split_by_sentence(block, NAVER_REQUEST_SIZE))

    print(f"3,000자 작업 블록: {len(work_blocks)}개")
    print(f"네이버 300자 이하 전송 블록: {len(naver_chunks)}개")
    print("전송 블록 글자 수:", [len(chunk) for chunk in naver_chunks])

    open_naver_spell_checker(driver)

    corrected_chunks = []
    for index, chunk in enumerate(naver_chunks, start=1):
        print(f"맞춤법 검사 중... {index}/{len(naver_chunks)}")
        corrected_chunks.append(spellcheck_one_chunk(driver, chunk))

    return "\n".join(corrected_chunks)


# -----------------------------------------------------------------------------
# 5. Papago 한국어 → 영어 번역
# -----------------------------------------------------------------------------
def open_papago(driver):
    driver.get("https://papago.naver.com/?sk=ko&tk=en")
    wait_visible(driver, PAPAGO_SOURCE_LOCATORS)
    wait_visible(driver, PAPAGO_TARGET_LOCATORS)


def translate_one_chunk(driver, text):
    source = wait_visible(driver, PAPAGO_SOURCE_LOCATORS)
    source.click()
    source.send_keys(Keys.CONTROL, "a")
    source.send_keys(Keys.BACKSPACE)

    # 이전 번역 결과가 사라진 것을 확인한 후 새 문장을 입력한다.
    WebDriverWait(driver, WAIT_SECONDS).until(
        lambda current_driver: not read_visible_text(
            current_driver, PAPAGO_TARGET_LOCATORS
        )
    )

    source.send_keys(text)
    time.sleep(REQUEST_PAUSE_SECONDS)

    return WebDriverWait(driver, WAIT_SECONDS).until(
        lambda current_driver: read_visible_text(
            current_driver, PAPAGO_TARGET_LOCATORS
        )
        or False
    )


def translate_full_text(driver, corrected_text):
    papago_chunks = split_by_sentence(corrected_text, PAPAGO_BLOCK_SIZE)

    print(f"Papago 3,000자 이하 번역 블록: {len(papago_chunks)}개")
    print("번역 블록 글자 수:", [len(chunk) for chunk in papago_chunks])

    open_papago(driver)

    translated_chunks = []
    for index, chunk in enumerate(papago_chunks, start=1):
        print(f"영어 번역 중... {index}/{len(papago_chunks)}")
        translated_chunks.append(translate_one_chunk(driver, chunk))

    return "\n".join(translated_chunks)


# -----------------------------------------------------------------------------
# 6. 전체 실행: 입력 TXT → 맞춤법후 TXT → 영어 번역 TXT
# -----------------------------------------------------------------------------
def main():
    DATA_DIR.mkdir(parents=True, exist_ok=True)

    if not INPUT_PATH.is_file():
        raise FileNotFoundError(
            "입력 파일을 찾을 수 없습니다. 다음 위치를 확인하세요.\n"
            f"{INPUT_PATH}"
        )

    # BOM이 있는 UTF-8 파일도 읽을 수 있도록 utf-8-sig를 사용한다.
    with open(INPUT_PATH, "r", encoding="utf-8-sig") as fp:
        original_text = fp.read().strip()

    if not original_text:
        raise ValueError("입력 파일이 비어 있습니다.")

    print(f"입력 파일: {INPUT_PATH}")
    print(f"입력 글자 수: {len(original_text):,}자")

    driver = None
    try:
        options = webdriver.ChromeOptions()
        options.add_argument("--start-maximized")
        driver = webdriver.Chrome(options=options)

        corrected_text = spellcheck_full_text(driver, original_text)
        with open(KOREAN_OUTPUT_PATH, "w", encoding="utf-8") as fp:
            fp.write(corrected_text)
        print(f"맞춤법 검사 완료: {KOREAN_OUTPUT_PATH}")

        translated_text = translate_full_text(driver, corrected_text)
        with open(ENGLISH_OUTPUT_PATH, "w", encoding="utf-8") as fp:
            fp.write(translated_text)
        print(f"영어 번역 완료: {ENGLISH_OUTPUT_PATH}")

    finally:
        if driver is not None:
            driver.quit()


if __name__ == "__main__":
    main()
